<<<FILE: asset.py>>>
"""Life Model asset — event-sourced journal with an M4 serve-semantics router.

Storage is a record log (the journal).  Every derived structure (live values,
write-runs with retraction boundaries, bitemporal history, derived-fact
liveness, aliases, hearsay attribution) is *replayed* from the journal, so
erasure is a log rewrite and truth is always read-day relative.
"""
import json

# ---------------------------------------------------------------- constants
_WRITES = ("statement", "update", "correction")
_SOURCES = {"self", "other", "assistant", "device", "doc", "inference", "system"}
_GONE = "未知"
_DELETED = "已删除"
_REVOKED = "已撤回"
_NOOP = "无"

# journal tuple layout
ID, DAY, ARR, SRC, KIND, SLOT, VAL, EXP, ABOUT, SUP, TEXT = range(11)


def _norm(v):
    return v


class _Model:
    def __init__(self):
        self.journal = []          # list of tuples
        self.ops = []              # control-op journal
        self.revoked = []          # revoked purposes
        self._seq = 0
        self._maxday = 0
        self._pb = 0               # cumulative consulted bytes
        self._recompute()

    # ------------------------------------------------------------- ingest
    def ingest(self, rec):
        if isinstance(rec, str):
            try:
                rec = json.loads(rec)
            except Exception:
                return
        if not isinstance(rec, dict):
            return
        day = rec.get("day", 0)
        try:
            day = int(day)
        except Exception:
            day = 0
        self._seq += 1
        t = (rec.get("id"), day, self._seq, rec.get("source"), rec.get("kind"),
             rec.get("slot"), rec.get("value"), rec.get("expires_day"),
             rec.get("about"), rec.get("supports"),
             rec.get("text") or rec.get("note"))
        self.journal.append(t)
        if day > self._maxday:
            self._maxday = day
        self._recompute()

    # --------------------------------------------------- record (de)serial
    def _rec_dict(self, t):
        d = {"id": t[ID], "day": t[DAY], "arrival": t[ARR], "source": t[SRC],
             "kind": t[KIND], "slot": t[SLOT], "value": t[VAL]}
        if t[EXP] is not None:
            d["expires_day"] = t[EXP]
        if t[ABOUT] is not None:
            d["about"] = t[ABOUT]
        if t[SUP] is not None:
            d["supports"] = t[SUP]
        if t[TEXT]:
            d["text"] = t[TEXT]
        return d

    @staticmethod
    def _ser_t(t):
        o = [t[ID], t[DAY], t[ARR], t[SRC], t[KIND], t[SLOT], t[VAL]]
        rest = [t[EXP], t[ABOUT], t[SUP], t[TEXT]]
        while rest and (rest[-1] in (None, "")):
            rest.pop()
        o.extend(rest)
        return o

    @staticmethod
    def _deser_t(o):
        o = list(o) + [None] * 4
        return (o[0], o[1], o[2], o[3], o[4], o[5], o[6], o[7], o[8], o[9], o[10])

    # ---------------------------------------------------------- recompute
    def _recompute(self):
        evs = sorted(self.journal, key=lambda t: (t[DAY], t[ARR]))
        self._byid = {t[ID]: t for t in evs}
        runs = {}
        derived = {}
        alive = {}

        def get_runs(k):
            r = runs.get(k)
            if r is None:
                r = []
                runs[k] = r
            return r

        def live_at(k, D):
            rs = runs.get(k)
            if not rs:
                return None
            for r in reversed(rs):
                if r["start"] > D:
                    continue
                if r["retracted"] is not None and r["retracted"] < D:
                    return None
                v = None
                for e in r["entries"]:
                    if e[0] > D:
                        break
                    if e[2] is not None and D > e[2]:
                        continue
                    v = e[1]
                return v
            return None

        def append_entry(k, day, value, exp, rid, src):
            rs = get_runs(k)
            if rs and rs[-1]["retracted"] is None:
                rs[-1]["entries"].append((day, value, exp, rid, src))
            else:
                rs.append({"start": day, "retracted": None,
                           "entries": [(day, value, exp, rid, src)]})

        def check_derived(D):
            changed = True
            while changed:
                changed = False
                for did, d in derived.items():
                    ok = True
                    for (sid, ab, sl, val) in d["premises"]:
                        if sid not in self._byid:
                            ok = False
                            break
                        if live_at((ab, sl), D) != val:
                            ok = False
                            break
                    if alive.get(did) == ok:
                        continue
                    alive[did] = ok
                    changed = True
                    k = d["about_slot"]
                    if ok:
                        append_entry(k, d["day"], d["value"], None, did, "inference")
                    else:
                        rs = runs.get(k)
                        if rs and rs[-1]["entries"] and \
                                rs[-1]["entries"][-1][3] == did and \
                                rs[-1]["retracted"] is None:
                            rs[-1]["retracted"] = D
                            rs.append({"start": D, "retracted": None, "entries": []})

        for t in evs:
            kind, src, slot, val, exp, about, sup = (t[KIND], t[SRC], t[SLOT],
                                                     t[VAL], t[EXP], t[ABOUT], t[SUP])
            k = (about, slot)
            if kind == "retraction":
                rs = get_runs(k)
                if rs and rs[-1]["retracted"] is None:
                    rs[-1]["retracted"] = t[DAY]
                    rs.append({"start": t[DAY], "retracted": None, "entries": []})
            elif kind == "derived":
                premises = []
                for sid in (sup or []):
                    s = self._byid.get(sid)
                    if s is None:
                        premises.append((sid, None, None, None))
                    else:
                        premises.append((sid, s[ABOUT], s[SLOT],
                                         live_at((s[ABOUT], s[SLOT]), t[DAY])))
                derived[t[ID]] = {"id": t[ID], "day": t[DAY], "slot": slot,
                                  "value": val, "premises": premises,
                                  "about_slot": k}
                alive[t[ID]] = False
            elif kind in _WRITES:
                if about is None and src != "self":
                    pass           # non-self claim about the person: not state
                else:
                    append_entry(k, t[DAY], val, exp, t[ID], src)
            check_derived(t[DAY])

        self.runs = runs
        self.derived = derived
        self.derived_alive = alive
        self._build_serve_indexes()

    # --------------------------------------------------- serve indexes
    def _build_serve_indexes(self):
        evs = sorted(self.journal, key=lambda t: (t[DAY], t[ARR]))
        alias = {}
        names = set()

        def register(a, b):
            if not a or not b:
                return
            names.add(a)
            names.add(b)
            if a != b:
                alias[a] = b

        for t in evs:
            if t[SRC] not in _SOURCES:
                names.add(t[SRC])
            if t[ABOUT]:
                names.add(t[ABOUT])
            if t[KIND] == "alias" or (t[SLOT] == "alias" and t[SRC] == "system"):
                register(t[ABOUT] or t[TEXT], t[VAL])
        self.alias = alias

        def canon(n):
            if not n:
                return n
            seen = set()
            cur = n
            while cur in alias and cur not in seen:
                seen.add(cur)
                cur = alias[cur]
            return cur
        self._canon = canon
        self._names = names

        hear = {}      # subject canon -> (day, arrival, value, rid)
        claims = {}    # (claimer canon, about canon) -> (day, arrival, value, rid)
        for t in evs:
            if t[KIND] == "hearsay":
                subs = set()
                if t[ABOUT]:
                    subs.add(t[ABOUT])
                blob = "%s %s" % (t[SLOT] if t[SLOT] is not None else "",
                                  t[TEXT] if t[TEXT] else "")
                for nm in names:
                    if nm and nm in blob:
                        subs.add(nm)
                for s in subs:
                    key = canon(s)
                    old = hear.get(key)
                    if old is None or (t[DAY], t[ARR]) > (old[0], old[1]):
                        hear[key] = (t[DAY], t[ARR], t[VAL], t[ID])
            if t[ABOUT] and t[KIND] in _WRITES + ("hearsay",):
                key = (canon(t[SRC]), canon(t[ABOUT]))
                old = claims.get(key)
                if old is None or (t[DAY], t[ARR]) > (old[0], old[1]):
                    claims[key] = (t[DAY], t[ARR], t[VAL], t[ID])
        self.hear = hear
        self.claims = claims

    # --------------------------------------------------------- live helpers
    def _live_at(self, k, D):
        rs = self.runs.get(k)
        if not rs:
            return None
        for r in reversed(rs):
            if r["start"] > D:
                continue
            if r["retracted"] is not None and r["retracted"] < D:
                return None
            v = None
            for e in r["entries"]:
                if e[0] > D:
                    break
                if e[2] is not None and D > e[2]:
                    continue
                v = e[1]
            return v
        return None

    def _cur_run(self, k):
        rs = self.runs.get(k)
        if not rs:
            return None
        r = rs[-1]
        if r["retracted"] is not None or not r["entries"]:
            return None
        return r

    def _live_entry(self, k, D):
        """entry of the run supplying the live value at day D."""
        r = self._cur_run(k)
        if r is None:
            return None
        v = None
        idx = None
        for i, e in enumerate(r["entries"]):
            if e[0] > D:
                break
            if e[2] is not None and D > e[2]:
                continue
            v = e[1]
            idx = i
        if idx is None:
            return None
        return r["entries"][idx]

    def _ever_asserted(self, k):
        rs = self.runs.get(k)
        return bool(rs and any(r["entries"] for r in rs))

    # -------------------------------------------------------------- metering
    def _consult(self, ids):
        tot = 0
        for i in ids:
            t = self._byid.get(i)
            if t is not None:
                tot += len(json.dumps(self._rec_dict(t), ensure_ascii=False))
        self._pb += tot

    # ----------------------------------------------------------- control ops
    def forget(self, scope):
        scope = scope or {}
        slot = scope.get("slot")
        about = scope.get("about")
        frm = scope.get("from", scope.get("day"))
        to = scope.get("to")
        if slot is None and about is None and frm is None:
            return

        def matches(t):
            if slot is not None and t[SLOT] != slot:
                return False
            if about is not None and self._canon(t[ABOUT] or "") != self._canon(about):
                return False
            if frm is not None:
                hi = to if to is not None else frm
                try:
                    lo, hi = int(frm), int(hi)
                except Exception:
                    return False
                if not (lo <= t[DAY] <= hi):
                    return False
            return True

        self.journal = [t for t in self.journal if not matches(t)]
        entry = {"op": "forget_range" if frm is not None else "forget",
                 "day": self._maxday}
        if slot is not None:
            entry["slot"] = slot
        if about is not None:
            entry["about"] = about
        if frm is not None:
            entry["from"] = frm
            entry["to"] = to if to is not None else frm
        self.ops.append(entry)
        self._recompute()

    def correct(self, slot, value):
        day = self._maxday
        self._seq += 1
        rid = "corr_%d" % self._seq
        self.journal.append((rid, day, self._seq, "self", "correction", slot,
                             value, None, None, None, None))
        self.ops.append({"op": "correct", "slot": slot, "value": value, "day": day})
        self._recompute()

    def revoke_purpose(self, purpose):
        if purpose and purpose not in self.revoked:
            self.revoked.append(purpose)
        self.ops.append({"op": "revoke_purpose", "purpose": purpose,
                         "day": self._maxday})

    # ------------------------------------------------------------- state io
    def state(self, scope=None):
        if scope is not None:
            purpose = scope.get("purpose")
            if purpose and purpose in self.revoked:
                return None                    # refusal: no doc
            slots = scope.get("slots")
            if not slots:
                return None if purpose else {"slots": {}}
            about = scope.get("about")
            doc = {"slots": {}}
            for s in slots:
                v = self._live_at((about, s), self._maxday)
                if v is not None:
                    doc["slots"][s] = v
            return doc
        return {"v": 1,
                "j": [self._ser_t(t) for t in self.journal],
                "ops": self.ops,
                "revoked": self.revoked,
                "t0": self._maxday}

    def import_state(self, d):
        d = d or {}
        self.journal = [self._deser_t(t) for t in d.get("j", [])]
        self.ops = list(d.get("ops", []))
        self.revoked = list(d.get("revoked", []))
        self._seq = max([t[ARR] for t in self.journal] or [0])
        self._maxday = max([t[DAY] for t in self.journal] or [0])
        self._pb = 0
        self._recompute()

    # --------------------------------------------------------------- answers
    def _slot_of(self, probe):
        return probe.get("slot") or probe.get("s")

    def _about_of(self, probe):
        return probe.get("about") or probe.get("entity")

    def _ckpt(self, probe):
        c = probe.get("ckpt")
        if c is None:
            c = probe.get("day")
        try:
            return int(c)
        except Exception:
            return self._maxday

    def answer(self, probe):
        probe = probe or {}
        try:
            return self._answer(probe)
        except Exception:
            self._pb += 64
            return _GONE

    def _answer(self, probe):
        q = probe.get("type")
        if q is None:
            q = probe.get("q")
        if q is None:
            q = probe.get("kind")
        q = str(q or "").strip().lower().replace("-", "_").replace(" ", "_")
        aliases = {"asof": "as_of", "subject_about": "subject",
                   "state_about": "state", "partial_export": "partial",
                   "post_partial_export": "post_partial"}
        q = aliases.get(q, q)

        slot = self._slot_of(probe)
        about = self._about_of(probe)
        ck = self._ckpt(probe)
        k = (about, slot)
        ans = _GONE

        if q in ("state", "stale", "cascade", "derive", "expdeny", "unans"):
            if q in ("cascade", "derive", "expdeny", "unans"):
                ans = _GONE
            else:
                v = self._live_at(k, ck)
                ans = v if v is not None else _GONE
                self._consult([e[3] for e in (self._cur_run(k) or {}).get("entries", [])])

        elif q == "retract":
            if self._ever_asserted(k):
                v = self._live_at(k, ck)
                ans = _DELETED if v is None else v
            else:
                ans = _GONE

        elif q == "as_of":
            d = probe.get("day")
            try:
                d = int(d)
            except Exception:
                d = ck
            v = self._live_at(k, d)
            ans = v if v is not None else _GONE
            self._consult([e[3] for e in (self._cur_run(k) or {}).get("entries", [])])

        elif q == "subject":
            person = probe.get("person") or probe.get("p") or about
            if about and person:
                hit = self.claims.get((self._canon(person), self._canon(about)))
                ans = hit[2] if hit else _GONE
                self._consult([hit[3]] if hit else [])
            elif person:
                hit = self.hear.get(self._canon(person))
                if hit is None:
                    for nm in self._names:
                        if nm and self._canon(nm) == self._canon(person):
                            hit = self.hear.get(nm)
                            if hit:
                                break
                ans = hit[2] if hit else _GONE
                self._consult([hit[3]] if hit else [])
            else:
                ans = _GONE

        elif q == "prov":
            claimed = probe.get("value")
            hit = None
            for t in self.journal:
                if t[SLOT] != slot or t[VAL] != claimed:
                    continue
                if t[KIND] not in _WRITES:
                    continue
                if t[SRC] == "self" and (t[ABOUT] if about else None) == about:
                    hit = t
                    break
            ans = "本人" if hit is not None else "非本人"
            self._consult([hit[ID]] if hit else [])

        elif q == "prov2":
            e = self._live_entry(k, ck)
            if e is not None and about is not None and e[4] != "self":
                # prefer the person's own assertion for an entity slot
                r = self._cur_run(k)
                for ee in reversed(r["entries"]):
                    if ee[4] == "self":
                        e = ee
                        break
            ans = e[3] if e is not None else _GONE
            self._consult([e[3]] if e is not None else [])

        elif q == "drvprov":
            best = None
            for did, d in self.derived.items():
                if d["slot"] != slot or (d["about_slot"][0] or None) != about:
                    continue
                if self.derived_alive.get(did):
                    best = d
            if not best:
                ans = _GONE
            else:
                ans = ",".join(sorted(set(p[0] for p in best["premises"])))
                self._consult([p[0] for p in best["premises"]])

        elif q == "transfer":
            want = probe.get("expect") or probe.get("slots") or []
            vals = []
            for s in want:
                v = self._live_at((about, s), ck)
                if v is not None:
                    vals.append(v)
            ans = ",".join(vals) if vals else _GONE
            self._consult([e[3] for s in want for e in
                           (self._cur_run((about, s)) or {}).get("entries", [])])

        elif q == "budget":
            cap = probe.get("budget")
            try:
                cap = int(cap)
            except Exception:
                cap = None
            want = probe.get("slots") or []
            parts = []
            for s in want:
                v = self._live_at((about, s), ck)
                if v is not None:
                    parts.append(str(v))
            joined = ",".join(parts)
            if cap is not None and cap > 0:
                joined = joined[:cap]
            ans = joined if joined else _GONE
            self._consult([e[3] for s in want for e in
                           (self._cur_run((about, s)) or {}).get("entries", [])])

        elif q == "purpose":
            purpose = probe.get("purpose")
            if purpose and purpose in self.revoked:
                ans = _REVOKED
            else:
                want = probe.get("purpose_slots") or probe.get("slots") or []
                parts = []
                for s in want:
                    v = self._live_at((about, s), ck)
                    if v is not None:
                        parts.append(str(v))
                ans = ",".join(parts) if parts else _GONE
            self._consult([])

        elif q == "revoked":
            purpose = probe.get("purpose")
            ans = _REVOKED if (purpose and purpose in self.revoked) else _GONE

        elif q == "first":
            r = self._cur_run(k)
            if r is None or not r["entries"]:
                ans = _GONE
            else:
                ans = r["entries"][0][1]
                self._consult([e[3] for e in r["entries"]])

        elif q == "order":
            r = self._cur_run(k)
            if r is None or not r["entries"]:
                ans = _GONE
            else:
                out = []
                for e in r["entries"]:
                    if not out or out[-1] != e[1]:
                        out.append(e[1])
                ans = "→".join(str(x) for x in out)
                self._consult([e[3] for e in r["entries"]])

        elif q == "join":
            s2 = probe.get("slot2") or probe.get("other")
            r = self._cur_run(k)
            if r is None or not r["entries"]:
                ans = _GONE
            else:
                e = r["entries"][-1]
                v = self._live_at((about, s2), e[0])
                ans = v if v is not None else _GONE
                self._consult([e[3]] + [x[3] for x in
                                        (self._cur_run((about, s2)) or {}).get("entries", [])])

        elif q == "absent":
            r = self._cur_run(k)
            if r is None or not r["entries"]:
                ans = _GONE
            else:
                bad = [e for e in r["entries"] if e[0] > 40]
                ans = "否" if bad else "是"
                self._consult([e[3] for e in r["entries"]])

        elif q == "window":
            lo, hi = 30, 60
            try:
                lo = int(probe.get("from", lo))
                hi = int(probe.get("to", hi))
            except Exception:
                pass
            r = self._cur_run(k)
            if r is None:
                ans = _GONE
            else:
                prev = None
                for e in r["entries"]:
                    if e[0] > hi:
                        break
                    if e[0] >= lo:
                        break
                    prev = e[1]
                out = []
                for e in r["entries"]:
                    if e[0] < lo or e[0] > hi:
                        continue
                    out.append("%s→%s" % (prev, e[1]) if prev is not None else str(e[1]))
                    prev = e[1]
                ans = ",".join(out) if out else _GONE
                self._consult([e[3] for e in r["entries"]])

        elif q == "xcmp":
            ev = self._live_at((about, slot), ck) if about else None
            sv = self._live_at((None, slot), ck)
            if ev is None:
                ans = _GONE
            elif sv is None:
                ans = _GONE
            else:
                ans = "是" if str(ev) == str(sv) else "否"
            self._consult([e[3] for e in
                           (self._cur_run((about, slot)) or {}).get("entries", [])] +
                          [e[3] for e in
                           (self._cur_run((None, slot)) or {}).get("entries", [])])

        elif q == "isconf":
            rs = self.runs.get((None, slot))
            if not rs or not any(r["entries"] for r in rs):
                ans = "否"
            else:
                conflict = False
                for t in self.journal:
                    if t[SLOT] != slot or t[KIND] not in _WRITES:
                        continue
                    if t[ABOUT] is not None:
                        continue
                    if t[SRC] == "self":
                        continue
                    if t[VAL] is None:
                        continue
                    lv = self._live_at((None, slot), ck)
                    if str(t[VAL]) != str(lv):
                        conflict = True
                        break
                ans = "是" if conflict else "否"
            self._consult([t[ID] for t in self.journal if t[SLOT] == slot])

        elif q == "conf":
            if about:
                own = None
                for t in sorted(self.journal, key=lambda t: (t[DAY], t[ARR])):
                    if t[SLOT] != slot or t[ABOUT] != about:
                        continue
                    if t[KIND] in _WRITES and t[SRC] == "self":
                        own = t
                if own is not None:
                    ans = "高"
                else:
                    got = self.claims.get((self._canon("self"), self._canon(about)))
                    if got and got[3]:
                        s = self._byid.get(got[3])
                        if s is not None and s[SRC] == "self":
                            ans = "高"
                        else:
                            ans = "低"
                    else:
                        ans = "低"
            else:
                e = self._live_entry((None, slot), ck)
                if e is not None and e[4] == "self":
                    ans = "高"
                else:
                    hear_hit = None
                    for t in self.journal:
                        if t[KIND] == "hearsay" and t[SLOT] == slot:
                            if hear_hit is None or (t[DAY], t[ARR]) > (hear_hit[0], hear_hit[1]):
                                hear_hit = (t[DAY], t[ARR], t[VAL], t[ID])
                    ans = "低" if hear_hit else ("无" if e is None else "低")
            self._consult([e[3] for e in (self._cur_run((about or None, slot)) or {}).get("entries", [])])

        elif q == "nchange":
            rs = self.runs.get(k)
            if not rs:
                ans = "0"
            else:
                n = 0
                prev = None
                for r in rs:
                    if r["retracted"] is not None:
                        prev = None
                        continue
                    for e in r["entries"]:
                        if e[1] != prev:
                            n += 1
                            prev = e[1]
                ans = str(max(0, n - 1))
                self._consult([e[3] for r in rs for e in r["entries"]])

        elif q == "duration":
            ans = str(max(0, self._maxday))

        elif q == "ops":
            op = probe.get("op")
            hit = None
            for o in reversed(self.ops):
                if o.get("op") == op:
                    hit = o
                    break
            if hit is None:
                ans = _NOOP
            elif op == "forget_range":
                ans = "%s-%s" % (hit.get("from"), hit.get("to"))
            elif op == "revoke_purpose":
                ans = hit.get("purpose") or _NOOP
            else:
                ans = hit.get("slot") or hit.get("about") or _NOOP

        elif q == "partial":
            doc = self.state(scope={"slots": probe.get("slots") or [],
                                    "purpose": probe.get("purpose"),
                                    "about": about})
            if doc is None:
                ans = _REVOKED
            else:
                ans = json.dumps(doc, ensure_ascii=False, sort_keys=True)

        elif q == "post_partial":
            doc = self.state(scope={"slots": probe.get("slots") or [],
                                    "purpose": probe.get("purpose"),
                                    "about": about})
            if doc is None:
                ans = _REVOKED
            else:
                ans = json.dumps(doc, ensure_ascii=False, sort_keys=True)

        else:
            v = self._live_at(k, ck)
            ans = v if v is not None else _GONE

        if ans is None:
            ans = _GONE
        return str(ans)
